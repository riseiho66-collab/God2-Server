cmake_minimum_required(VERSION 3.24)
project(God2NativeLab LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "God2NativeLab requires a separate Windows Win32 build")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601 UNICODE _UNICODE)
add_library(God2LabAgent SHARED agent.cpp agent.def)
target_link_libraries(God2LabAgent PRIVATE bcrypt)
get_filename_component(LAB_COPY "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/capture_work/host-20260918-1430/client/God2.exe" ABSOLUTE)
target_compile_definitions(God2LabAgent PRIVATE GOD2_LAB_WORKING_COPY=L"${LAB_COPY}")
add_executable(God2LabAgentGuardTest agent_test.cpp)
include(CTest)
add_test(NAME native_lab_agent_refusal COMMAND God2LabAgentGuardTest $<TARGET_FILE:God2LabAgent>)
add_library(God2NativeTransaction STATIC transaction.cpp)
add_executable(God2NativeTransactionTest transaction_test.cpp)
target_link_libraries(God2NativeTransactionTest PRIVATE God2NativeTransaction)
add_test(NAME native_transaction_synthetic COMMAND God2NativeTransactionTest)
