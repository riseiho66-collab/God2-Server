cmake_minimum_required(VERSION 3.20)
project(god2_tw_map_objects LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This offline tool uses Windows BCrypt for SHA-256.")
endif()
add_executable(god2-tw-map-objects tw_board_patch.cpp)
target_compile_features(god2-tw-map-objects PRIVATE cxx_std_20)
target_link_libraries(god2-tw-map-objects PRIVATE bcrypt)
add_executable(god2-tw-town-door-can tw_door_can_patch.cpp)
target_compile_features(god2-tw-town-door-can PRIVATE cxx_std_20)
target_link_libraries(god2-tw-town-door-can PRIVATE bcrypt)
if(MSVC)
  target_compile_options(god2-tw-map-objects PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(god2-tw-town-door-can PRIVATE /W4 /permissive- /utf-8)
endif()
