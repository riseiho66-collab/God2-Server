cmake_minimum_required(VERSION 3.20)
project(god2_tw_client_compat LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The offline staging tool uses Windows BCrypt.")
endif()
add_executable(god2-tw-name-card-guard tw_name_card_guard.cpp)
target_compile_features(god2-tw-name-card-guard PRIVATE cxx_std_20)
target_link_libraries(god2-tw-name-card-guard PRIVATE bcrypt)
if(MSVC)
  target_compile_options(god2-tw-name-card-guard PRIVATE /W4 /permissive- /utf-8)
endif()
