cmake_minimum_required(VERSION 3.24)
project(God2LocalCryptoOracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GOD2_ENABLE_ORACLE_LAB "Enable non-distributable local oracle tools" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(God2CryptoClean SHARED clean_bridge.cpp ../../src/crypto/God2CryptoClean.cpp)
target_include_directories(God2CryptoClean PRIVATE ../../src)
target_compile_definitions(God2CryptoClean PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(God2CryptoClean PRIVATE /W4 /permissive- /utf-8)
endif()
if(GOD2_ENABLE_ORACLE_LAB)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../generated/crypto_candidate.cpp")
    add_library(God2CryptoCandidate SHARED ../../generated/crypto_candidate.cpp)
  endif()
  foreach(tool IN ITEMS God2CryptoOracle god2-crypto-diff)
    add_executable(${tool} launcher.cpp)
    if(tool STREQUAL "God2CryptoOracle")
      set(script oracle.py)
    else()
      set(script differential.py)
    endif()
    target_compile_definitions(${tool} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
      ORACLE_PYTHON=L"${Python3_EXECUTABLE}" ORACLE_SCRIPT=L"${CMAKE_CURRENT_SOURCE_DIR}/${script}")
  endforeach()
endif()
# Deliberately no install/export/package rule for oracle tools or captured code.
