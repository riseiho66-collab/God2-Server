cmake_minimum_required(VERSION 3.24)
project(God2StrikeLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(tool IN ITEMS God2LabLauncher God2SnapshotDiff God2LoopbackProbe)
  add_executable(${tool} frontend.cpp)
  target_compile_definitions(${tool} PRIVATE UNICODE _UNICODE LAB_TOOL=L"${tool}" LAB_PYTHON=L"${Python3_EXECUTABLE}" LAB_SCRIPT=L"${CMAKE_CURRENT_SOURCE_DIR}/lab.py")
endforeach()
add_library(God2Probe SHARED probe.cpp packet_probe.cpp)
target_include_directories(God2Probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/_deps/json-src/include")
target_compile_definitions(God2Probe PRIVATE UNICODE _UNICODE)
target_link_libraries(God2Probe PRIVATE bcrypt)
add_library(God2Compat SHARED compat.cpp)
target_compile_definitions(God2Compat PRIVATE UNICODE _UNICODE)
add_executable(God2PrivateDesktop private_desktop_runner.cpp)
target_compile_definitions(God2PrivateDesktop PRIVATE UNICODE _UNICODE)
target_link_libraries(God2PrivateDesktop PRIVATE user32)
add_executable(God2CompatSemantics compat_test.cpp)
target_link_libraries(God2CompatSemantics PRIVATE winmm)
foreach(tool IN ITEMS God2CompatHost god2-strike-runner)
  add_executable(${tool} frontend.cpp native_dialog.cpp)
  target_include_directories(${tool} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/_deps/json-src/include")
  target_compile_definitions(${tool} PRIVATE NATIVE_DIALOG)
  target_link_libraries(${tool} PRIVATE user32)
  target_compile_definitions(${tool} PRIVATE UNICODE _UNICODE LAB_TOOL=L"${tool}" LAB_PYTHON=L"${Python3_EXECUTABLE}" LAB_SCRIPT=L"${CMAKE_CURRENT_SOURCE_DIR}/apex.py")
endforeach()
