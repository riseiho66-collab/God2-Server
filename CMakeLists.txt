cmake_minimum_required(VERSION 3.24)
project(God2Rebuild VERSION 0.1.0 LANGUAGES C CXX)
option(GOD2_ENABLE_ORACLE_LAB "Oracle backend is forbidden in production server targets" OFF)
if(GOD2_ENABLE_ORACLE_LAB)
  message(FATAL_ERROR "OracleLab is local research only. Build tools/crypto_oracle separately; production server forbids it.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(cmake/Dependencies.cmake)
file(GLOB_RECURSE GOD2_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER GOD2_SOURCES EXCLUDE REGEX "/(main|replay_main)\\.cpp$")
add_library(god2_core STATIC ${GOD2_SOURCES})
target_include_directories(god2_core PUBLIC src)
target_link_libraries(god2_core PUBLIC asio spdlog::spdlog nlohmann_json::nlohmann_json sol2::sol2 lua sqlite3)
if(WIN32)
  target_link_libraries(god2_core PRIVATE bcrypt shell32)
endif()
target_compile_definitions(god2_core PUBLIC SOL_ALL_SAFETIES_ON=1)
if(MSVC)
  target_compile_options(god2_core PRIVATE /W4 /permissive- /bigobj /MP)
  target_compile_options(god2_core PUBLIC /utf-8)
  target_compile_definitions(god2_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
else()
  target_compile_options(god2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(god2-server src/main.cpp)
target_link_libraries(god2-server PRIVATE god2_core)
add_executable(god2-server-fruit EXCLUDE_FROM_ALL src/main.cpp)
target_link_libraries(god2-server-fruit PRIVATE god2_core)
add_executable(god2-server-native-results EXCLUDE_FROM_ALL src/main.cpp)
target_link_libraries(god2-server-native-results PRIVATE god2_core)
add_executable(protocol-replay src/tools/replay_main.cpp)
target_link_libraries(protocol-replay PRIVATE god2_core)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS god2-server protocol-replay RUNTIME DESTINATION .)
install(DIRECTORY config scripts resources DESTINATION .)
